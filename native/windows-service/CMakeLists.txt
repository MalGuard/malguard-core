cmake_minimum_required(VERSION 3.20)
project(MalGuardService LANGUAGES CXX)
add_executable(MalGuardService MalGuardService.cpp)
target_compile_features(MalGuardService PRIVATE cxx_std_17)
target_compile_definitions(MalGuardService PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(MalGuardService PRIVATE advapi32)
if(MSVC)
  target_compile_options(MalGuardService PRIVATE /W4 /WX /guard:cf /sdl)
  target_link_options(MalGuardService PRIVATE /DYNAMICBASE /NXCOMPAT /HIGHENTROPYVA)
endif()
