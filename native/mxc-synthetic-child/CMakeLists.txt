cmake_minimum_required(VERSION 3.20)
project(MalGuardMxcSyntheticChild LANGUAGES CXX)
add_executable(MalGuardMxcSyntheticChild MalGuardMxcSyntheticChild.cpp)
target_compile_features(MalGuardMxcSyntheticChild PRIVATE cxx_std_17)
target_compile_definitions(MalGuardMxcSyntheticChild PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX WINVER=0x0A00 _WIN32_WINNT=0x0A00)
target_link_libraries(MalGuardMxcSyntheticChild PRIVATE Ws2_32)
if(MSVC)
  set_property(TARGET MalGuardMxcSyntheticChild PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  target_compile_options(MalGuardMxcSyntheticChild PRIVATE /W4 /WX /guard:cf /sdl)
  target_link_options(MalGuardMxcSyntheticChild PRIVATE /DYNAMICBASE /NXCOMPAT /HIGHENTROPYVA /guard:cf)
endif()
