cmake_minimum_required(VERSION 3.20)
project(MalGuardInstaller LANGUAGES CXX RC)
if(NOT DEFINED PAYLOAD_PATH OR NOT DEFINED INSTALL_SCRIPT_PATH OR NOT DEFINED PAYLOAD_SHA256 OR NOT DEFINED INSTALL_SCRIPT_SHA256)
  message(FATAL_ERROR "Installer resource paths and hashes are required")
endif()
configure_file(resources.rc.in resources.rc @ONLY)
add_executable(MalGuardSetup WIN32 main.cpp "${CMAKE_CURRENT_BINARY_DIR}/resources.rc")
target_compile_features(MalGuardSetup PRIVATE cxx_std_17)
target_compile_definitions(MalGuardSetup PRIVATE PAYLOAD_SHA256="${PAYLOAD_SHA256}" INSTALL_SCRIPT_SHA256="${INSTALL_SCRIPT_SHA256}")
target_link_libraries(MalGuardSetup PRIVATE bcrypt)
